4.1.1 Plot simulated beaker length versus age using MM1_lme model

Prepare data to create the plot, calculate SE

MyData <- data.frame(Age = seq(min(WS2$Age), 
                          max(WS2$Age),
                          length = 25))
X <- model.matrix(~ Age, data = MyData)
MyData$Pred <- X %*% fixef(MM1_lme)
MyData$SE <- sqrt(diag(X %*% vcov(MM1_lme) %*% t(X))  )
MyData$SeUp <- MyData$Pred + 1.96 * MyData$SE
MyData$SeLo <- MyData$Pred - 1.96 * MyData$SE

Create plot

ggplot() + 
  geom_point(data = WS2, aes(x = Age, y = Beak), shape = 1, size = 1) +
  geom_line(data = MyData, aes(x = Age, y = Pred), colour = "black") +
  geom_ribbon(data = MyData, aes(x = Age, ymin = SeLo, ymax = SeUp), alpha = 0.6) + # add ribbin of SE
  labs(x = "Age (days)",
    y = "Beak length (mm)") +
  theme(text = element_text(size = 12))
